306 BOOL CPaneHost::CreateFromTemplate(UINT nIDTemplate, CWnd* pParent)
307 {
308     HINSTANCE hInst = AfxFindResourceHandle(MAKEINTRESOURCE(nIDTemplate), RT_DIALOG);
309     HRSRC hRes = ::FindResource(hInst, MAKEINTRESOURCE(nIDTemplate), RT_DIALOG);
310     if (hRes == NULL)
311         return FALSE;
312     HGLOBAL hMem = ::LoadResource(hInst, hRes);
313     if (hMem == NULL)
314         return FALSE;
315     LPCDLGTEMPLATE pTpl = (LPCDLGTEMPLATE)::LockResource(hMem);
316     if (pTpl == NULL)
317         return FALSE;
318 
319     HWND hWnd = ::CreateDialogIndirect(hInst, pTpl, pParent->GetSafeHwnd(), NULL);
320     if (hWnd == NULL)
321         return FALSE;
322     if (!SubclassWindow(hWnd))
323     {
324         ::DestroyWindow(hWnd);
325         return FALSE;
326     }
327     return TRUE;
328 }

330 // 通知(WM_NOTIFY/WM_COMMAND)是发给"控件的父窗口"的；控件挂在 #32770 下，
331 // 所以要转给 pane，pane 的 ON_NOTIFY(TCN_SELCHANGE) 才收得到。
332 // ⚠️ **WM_DRAWITEM 也一样**：owner-draw 的按钮把画自己的请求发给**直接父窗口**（#32770），
333 //    不转给 pane 的话 `ON_WM_DRAWITEM` 永远不会被调到 —— 现象是"按钮在窗口树里可见、
334 //    尺寸也对，就是一片灰什么都不画"（本轮踩过）。
335 LRESULT CPaneHost::WindowProc(UINT message, WPARAM wParam, LPARAM lParam)
336 {
337     if (message == WM_NOTIFY || message == WM_COMMAND ||
338         message == WM_DRAWITEM || message == WM_MEASUREITEM || message == WM_COMPAREITEM)
339     {
340         CWnd* pPane = GetParent();
341         if (pPane != NULL && pPane->GetSafeHwnd() != NULL)
342             return pPane->SendMessage(message, wParam, lParam);
343     }
344     return CWnd::WindowProc(message, wParam, lParam);
