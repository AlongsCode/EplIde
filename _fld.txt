2127 static CString EcDeclName(const CString& line, LPCTSTR pszHead)
2128 {
2129     // line 形如 ".子程序 加法, 整数型" / ".程序集 启动窗口程序集"，返回逗号/空格后的第一个词
2130     CString rest = line.Mid(1);
2131     CString head = rest;
2132     int sp = -1;
2133     for (int i = 0; i < rest.GetLength(); i++)
2134     {
2135         unsigned char c = (unsigned char)rest[i];
2136         if (c == ' ' || c == '\t') { sp = i; break; }
2137     }
2138     if (sp < 0) return _T("");
2139     head = rest.Left(sp);
2140     if (head != pszHead) return _T("");
2141     CString body = rest.Mid(sp);
2142     body.TrimLeft();
2143     int e = 0;
2144     while (e < body.GetLength() && body[e] != _T(',') && (unsigned char)body[e] != 0xA3) e++;
2145     CString name = body.Left(e);
2146     name.TrimRight();
2147     return name;
2148 }
